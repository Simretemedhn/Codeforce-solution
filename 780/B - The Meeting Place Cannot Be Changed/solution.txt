n = int(input())
positions = list(map(int, input().split()))
speeds = list(map(int, input().split()))
 
def max_time(meeting_point):
    max_t = 0
    for i in range(n):
        time_needed = abs(positions[i] - meeting_point) / speeds[i]
        max_t = max(max_t, time_needed)
    return max_t
 
# Ternary search on the meeting point
left = min(positions)
right = max(positions)
 
for _ in range(200):  # Enough iterations for high precision
    mid1 = left + (right - left) / 3
    mid2 = right - (right - left) / 3
    
    if max_time(mid1) < max_time(mid2):
        right = mid2
    else:
        left = mid1
 
# Check the best point
meeting_point = (left + right) / 2
print(f"{max_time(meeting_point):.12f}")
