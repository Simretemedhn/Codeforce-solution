t = int(input())
 
for _ in range(t):
    n = int(input())
    list1 = list(map(int, input().split()))
    
    if n == 1:
        if list1[0] < 3:
            print("NO")
        else:
            print("YES")
    else:
        if max(list1) >= 3:
            print("YES")
        else:
            count = 0 
            for i in range(len(list1)):
                if list1[i] >= 2:
                    count += 1 
            print("YES" if count >= 2 else "NO")
