t = int(input())
 
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
 
    mismatch_count = 0 
    count_ones = 0 
    for i in range(n):
        if a[i] != b[i]:
            mismatch_count += 1 
            if a[i] == 1:
                count_ones += 1 
                
    if mismatch_count == 0:
        print(0)
    elif sum(a) == 0: 
        print(-1)
    elif count_ones % 2 != 0:  
        print(1)
    else:  
        if count_ones == 0 and sum(b) == n:  
            print(-1)
        else:
            print(2)
